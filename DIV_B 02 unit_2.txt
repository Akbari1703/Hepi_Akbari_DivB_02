 30      IF monthly_salary < 25000 THEN
 31          decision := 'REJECTED';
 32
 33      -- Maximum Loan Check
 34      ELSIF loan_amount > (monthly_salary * 60) THEN
 35          decision := 'REJECTED';
 36
 37      ELSE
 38
 39          -- EMI Formula
 40          emi := approved_amount * monthly_rate *
 41                 POWER(1 + monthly_rate, tenure_months) /
 42                 (POWER(1 + monthly_rate, tenure_months) - 1);
 43
 44          -- Reduce Loan Until EMI Fits FOIR
 45          WHILE emi > available_emi LOOP
 46
 47              approved_amount := approved_amount - 10000;
 48
 49              IF approved_amount <= 0 THEN
 50                  EXIT;
 51              END IF;
 52
 53              emi := approved_amount * monthly_rate *
 54                     POWER(1 + monthly_rate, tenure_months) /
 55                     (POWER(1 + monthly_rate, tenure_months) - 1);
 56
 57          END LOOP;
 58
 59          -- Decision using CASE
 60          decision :=
 61          CASE
 62              WHEN approved_amount = loan_amount THEN
 63                  'APPROVED'
 64              WHEN approved_amount > 0 THEN
 65                  'CONDITIONAL'
 66              ELSE
 67                  'REJECTED'
 68          END;
 69      END IF;
 70
 71      -- Output
 72      DBMS_OUTPUT.PUT_LINE('========== LOAN REPORT ==========');
 73      DBMS_OUTPUT.PUT_LINE('Monthly Salary      : Rs. ' || monthly_salary);
 74      DBMS_OUTPUT.PUT_LINE('Existing EMI        : Rs. ' || existing_emi);
 75      DBMS_OUTPUT.PUT_LINE('Requested Loan      : Rs. ' || loan_amount);
 76      DBMS_OUTPUT.PUT_LINE('Interest Rate       : ' || annual_rate || '%');
 77      DBMS_OUTPUT.PUT_LINE('Tenure              : ' || tenure_months || ' Months');
 78      DBMS_OUTPUT.PUT_LINE('---------------------------------');
 79      DBMS_OUTPUT.PUT_LINE('FOIR Limit (40%)    : Rs. ' || ROUND(foir_limit));
 80      DBMS_OUTPUT.PUT_LINE('Available EMI       : Rs. ' || ROUND(available_emi));
 81
 82      IF approved_amount > 0 THEN
 83          DBMS_OUTPUT.PUT_LINE('Calculated EMI      : Rs. ' || ROUND(emi));
 84      ELSE
 85          DBMS_OUTPUT.PUT_LINE('Calculated EMI      : Not Applicable');
 86      END IF;
 87
 88      DBMS_OUTPUT.PUT_LINE('Approved Loan       : Rs. ' || approved_amount);
 89      DBMS_OUTPUT.PUT_LINE('Decision            : ' || decision);
 90
 91  END;
